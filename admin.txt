let slugTimeout;

document.addEventListener('DOMContentLoaded', function() {
  const titleInput = document.getElementById('title');
  const slugInput = document.getElementById('slug');
  const slugPreview = document.getElementById('slugPreview');
  const imageUrlInput = document.getElementById('imageUrl');
  const imagePreview = document.getElementById('imagePreview');
  const previewImg = document.getElementById('previewImg');

  // Title counter
  titleInput.addEventListener('input', function() {
    document.getElementById('titleCount').textContent = this.value.length + ' / 200 characters';
    
    // Auto-generate slug
    clearTimeout(slugTimeout);
    slugTimeout = setTimeout(() => {
      if (!slugInput.value || slugInput.dataset.autoGenerated !== 'false') {
        const slug = this.value
          .toLowerCase()
          .replace(/[^a-z0-9]+/g, '-')
          .replace(/^-+|-+$/g, '');
        slugInput.value = slug;
        slugInput.dataset.autoGenerated = 'true';
        slugPreview.textContent = slug || 'your-slug';
      }
    }, 300);
  });

  // Slug manual edit
  slugInput.addEventListener('input', function() {
    this.dataset.autoGenerated = 'false';
    slugPreview.textContent = this.value || 'your-slug';
  });

  // Excerpt counter
  document.getElementById('excerpt').addEventListener('input', function() {
    document.getElementById('excerptCount').textContent = this.value.length + ' / 500 characters';
  });

  // Content counter
  document.getElementById('content').addEventListener('input', function() {
    document.getElementById('contentCount').textContent = this.value.length + ' characters';
  });

  // Meta description counter
  document.getElementById('metaDescription').addEventListener('input', function() {
    document.getElementById('metaDescCount').textContent = this.value.length + ' / 200 characters';
  });

  // Image preview
  imageUrlInput.addEventListener('input', function() {
    const url = this.value.trim();
    if (url && (url.startsWith('http://') || url.startsWith('https://'))) {
      previewImg.src = url;
      imagePreview.style.display = 'block';
    } else {
      imagePreview.style.display = 'none';
    }
  });
});

async function handleSubmit(e) {
  e.preventDefault();
  
  const form = document.getElementById('createForm');
  const submitBtn = document.getElementById('submitBtn');
  const notification = document.getElementById('notification');
  
  const formData = new FormData(form);
  const data = {
    title: formData.get('title').trim(),
    slug: formData.get('slug').trim(),
    excerpt: formData.get('excerpt') || null,
    content: formData.get('content').trim(),
    imageUrl: formData.get('imageUrl') || null,
    metaDescription: formData.get('metaDescription') || null,
    metaKeywords: formData.get('metaKeywords') || null,
    isPublished: formData.get('isPublished') === 'on',
  };

  // Validation
  if (!data.title) {
    showNotification('❌ Title is required', 'error');
    return;
  }
  if (!data.slug) {
    showNotification('❌ Slug is required', 'error');
    return;
  }
  if (!data.content) {
    showNotification('❌ Content is required', 'error');
    return;
  }

  submitBtn.textContent = '⏳ Creating...';
  submitBtn.disabled = true;
  notification.style.display = 'none';

  try {
    const response = await fetch('/api/admin/blog', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(data),
      credentials: 'include',
    });

    const result = await response.json();

    if (result.success) {
      showNotification('✅ Resource created successfully!', 'success');
      setTimeout(() => {
        window.location.href = '/admin/blog';
      }, 1500);
    } else {
      showNotification('❌ Error: ' + (result.error || 'Failed to create resource'), 'error');
      submitBtn.textContent = '📝 Create Resource';
      submitBtn.disabled = false;
    }
  } catch (error) {
    showNotification('❌ Network error: ' + error.message, 'error');
    submitBtn.textContent = '📝 Create Resource';
    submitBtn.disabled = false;
  }
}

function showNotification(message, type) {
  const el = document.getElementById('notification');
  el.className = type;
  el.textContent = message;
  el.style.display = 'block';
}

async function handleSubmit(e) {
  e.preventDefault();
  
  const form = document.getElementById('updateForm');
  const submitBtn = document.getElementById('submitBtn');
  const notification = document.getElementById('notification');
  
  const formData = new FormData(form);
  const data = {
    title: formData.get('title').trim(),
    slug: formData.get('slug').trim(),
    excerpt: formData.get('excerpt') || null,
    content: formData.get('content').trim(),
    imageUrl: formData.get('imageUrl') || null,
    metaDescription: formData.get('metaDescription') || null,
    metaKeywords: formData.get('metaKeywords') || null,
    isPublished: formData.get('isPublished') === 'on',
  };

  // Validation
  if (!data.title) { showNotification('❌ Title is required', 'error'); return; }
  if (!data.slug) { showNotification('❌ Slug is required', 'error'); return; }
  if (!data.content) { showNotification('❌ Content is required', 'error'); return; }

  submitBtn.textContent = '⏳ Updating...';
  submitBtn.disabled = true;
  notification.style.display = 'none';

  try {
    const response = await fetch(`/api/admin/blog/${postId}`, {
      method: 'PUT',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(data),
      credentials: 'include',
    });

    const result = await response.json();

    if (result.success) {
      showNotification('✅ Resource updated successfully!', 'success');
      setTimeout(() => {
        window.location.href = '/admin/blog';

      showNotification('❌ Error: ' + (result.error || 'Failed to update resource'), 'error');
      submitBtn.textContent = '💾 Update Resource';
      submitBtn.disabled = false;
    }
  } catch (error) {

  }
}

function filterPosts(filter) {

function renderPosts() {
  const filtered = allPosts.filter(post => {
    if (currentFilter === 'published') return post.isPublished;
    if (currentFilter === 'draft') return !post.isPublished;
    return true;
  });

  const tableBody = document.getElementById('postsTableBody');
  document.getElementById('postCount').textContent = filtered.length + ' posts';

  if (filtered.length === 0) {
    tableBody.innerHTML = `
      <tr><td colspan="4" style="padding: 2rem; text-align: center; color: #999;">
        ${currentFilter === 'all' ? 'No posts found.' : 'No ' + currentFilter + ' posts found.'}
      </td></tr>
    `;
    return;
  }

  tableBody.innerHTML = filtered.map(post => `
    <tr class="table-row">
      <td style="padding: 1rem;">
        <strong style="color: #4A3528;">${escapeHtml(post.title)}</strong>
      </td>
      <td style="padding: 1rem;">
        <span class="status-badge ${post.isPublished ? 'status-published' : 'status-draft'}">
          ${post.isPublished ? '✅ Published' : '📝 Draft'}
        </span>
      </td>
      <td style="padding: 1rem; color: #666; font-size: 0.9rem;">
        ${formatDate(post.createdAt)}
      </td>
      <td style="padding: 0.5rem;">
        <div class="action-group">
          ${post.isPublished 
            ? `<button onclick="togglePublish('${post.id}', false)" class="btn-sm btn-warning" title="Unpublish">📤 Unpublish</button>`
            : `<button onclick="togglePublish('${post.id}', true)" class="btn-sm btn-success" title="Publish">📢 Publish</button>`
          }
          <a href="/admin/blog/edit/${post.id}" class="btn-sm btn-primary" style="text-decoration: none;">✏️ Edit</a>

  `).join('');

function escapeHtml(text) {
  if (!text) return '';
  const div = document.createElement('div');
  div.textContent = text;
  return div.innerHTML;
}

function formatDate(dateStr) {
  if (!dateStr) return 'N/A';
  const d = new Date(dateStr);
  return d.toLocaleDateString('en-US', { year: 'numeric', month: 'short', day: 'numeric' });
}
